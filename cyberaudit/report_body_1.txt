def build_html_report(
    session_id: str,
    target: str,
    open_ports: list[dict],
    vulns: list[dict],
    hosts_up: list[str],
) -> str:
    score, band = risk_score(vulns)
    esc = html.escape
    rows = []
    order = {"critical": 0, "high": 1, "medium": 2, "low": 3}
    for v in sorted(vulns, key=lambda x: order.get(x.get("severity") or "", 9)):
        plain = _plain_for(v.get("service") or "", v.get("title") or "")
        rows.append(
            f"<tr><td>{esc(severity_label(v.get('severity') or ''))}</td>"
            f"<td>{esc(str(v.get('host')))}:{esc(str(v.get('port')))}</td>"
            f"<td><strong>{esc(plain['what'])}</strong><br>"
            f"<span class='muted'>{esc(plain['why'])}</span><br>"
            f"<em>Action: {esc(plain['do'])}</em></td>"
            f"<td>{esc(str(v.get('cma_section') or '—'))}</td></tr>"
        )
    host_lis = "".join(
        f"<li><code>{esc(h)}</code> — "
        + esc(
            ", ".join(
                f"{p.get('port')}/{p.get('service')}"
                for p in open_ports
                if p.get("host") == h
            )[:120]
            or "no common ports"
        )
        + "</li>"
        for h in hosts_up
    ) or "<li>No hosts responded</li>"

    return f"""<!DOCTYPE html>
<html lang="en"><head><meta charset="utf-8">
<title>CyberAudit Report {esc(session_id)}</title>
<style>
body{{font-family:system-ui,Segoe UI,sans-serif;max-width:900px;margin:2rem auto;padding:0 1rem;color:#0f172a;line-height:1.5}}
h1{{color:#0ea5e9}} .score{{font-size:1.5rem;font-weight:700}}
table{{width:100%;border-collapse:collapse;margin:1rem 0}}
th,td{{border:1px solid #cbd5e1;padding:.6rem;text-align:left;vertical-align:top}}
th{{background:#0f172a;color:#fff}} .muted{{color:#64748b;font-size:.9rem}}
.footer{{margin-top:2rem;font-size:.85rem;color:#64748b;border-top:1px solid #e2e8f0;padding-top:1rem}}
</style></head><body>
<h1>{esc(COMPANY['name'])}</h1>
<p><strong>Network Security Audit — Client Report</strong><br>
{esc(COMPANY['tagline'])}</p>
<p>Report ID: <code>{esc(session_id)}</code><br>
Date: {esc(datetime.datetime.now().strftime('%d %B %Y'))}<br>
Scope: <code>{esc(target)}</code></p>
<p class="score">Posture score: {score:.0f}/100 — {esc(band)}</p>
<p>Systems responding: <strong>{len(hosts_up)}</strong> ·
Open services: <strong>{len(open_ports)}</strong> ·
Issues: <strong>{len(vulns)}</strong></p>
<h2>Systems on the network</h2>
<ul>{host_lis}</ul>
<h2>Findings (plain language)</h2>
<table><thead><tr><th>Priority</th><th>Where</th><th>Explanation & action</th><th>CMA</th></tr></thead>
<tbody>{''.join(rows) if rows else '<tr><td colspan="4">No risk findings from this scan.</td></tr>'}</tbody></table>
<h2>What you should do next</h2>
<ol>
<li>Assign owners for Critical and High items.</li>
<li>Close unused services; restrict remote access behind VPN + MFA.</li>
<li>Verify backups and schedule a re-scan after fixes.</li>
</ol>
<div class="footer">
<p>Prepared by {esc(COMPANY['name'])} · {esc(COMPANY['email'])} · {esc(COMPANY['phone'])}<br>
{esc(COMPANY['address'])} · {esc(COMPANY['website'])}<br>
References TT Computer Misuse Act Chap. 11:17 for awareness — not legal advice.
Confidential to the engagement client.</p>
</div>
</body></html>
"""


def write_client_reports(
    session_id: str,
    target: str,
    open_ports: list[dict],
    vulns: list[dict],
    hosts_up: list[str],
) -> dict[str, Path]:
    config.REPORTS_DIR.mkdir(parents=True, exist_ok=True)
    text = build_executive_summary(session_id, target, open_ports, vulns, hosts_up)
    html_doc = build_html_report(session_id, target, open_ports, vulns, hosts_up)

    txt_path = config.REPORTS_DIR / f"CyberAudit_Client_{session_id}.txt"
    html_path = config.REPORTS_DIR / f"CyberAudit_Client_{session_id}.html"
    tech_path = config.REPORTS_DIR / f"CyberAudit_Technical_{session_id}.txt"

    txt_path.write_text(text, encoding="utf-8")
    html_path.write_text(html_doc, encoding="utf-8")

    tech_lines = [
        f"Technical annex — {session_id}",
        f"Target: {target}",
        f"Hosts: {', '.join(hosts_up)}",
        "",
        "Open ports:",
    ]
    for p in open_ports:
        tech_lines.append(f"  {p.get('host')}:{p.get('port')}/{p.get('service')} {p.get('banner','')}")
    tech_lines.append("")
    tech_lines.append("Findings:")
    for v in vulns:
        tech_lines.append(
            f"  [{v.get('severity')}] {v.get('host')}:{v.get('port')} "
            f"{v.get('title')} | CMA {v.get('cma_section')} | {v.get('cve','')}"
        )
    tech_path.write_text("\n".join(tech_lines) + "\n", encoding="utf-8")

    pdf_path = config.REPORTS_DIR / f"CyberAudit_Client_{session_id}.pdf"
    pdf_path.write_bytes(
        b"%PDF-1.4\n% CyberAudit client report - open the .html or .txt for full content\n"
        + text[:4000].encode("utf-8", errors="replace")
    )

    return {
        "executive_txt": txt_path,
        "executive_html": html_path,
        "technical_txt": tech_path,
        "pdf_stub": pdf_path,
    }
