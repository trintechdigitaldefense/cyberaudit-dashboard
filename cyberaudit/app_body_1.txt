@app.route("/login", methods=["GET", "POST"])
def login():
    if not config.AUTH_REQUIRED:
        return redirect(url_for("index"))
    error = None
    if request.method == "POST":
        user = request.form.get("username", "")
        password = request.form.get("password", "")
        role = None
        if (
            config.AUTH_USER
            and config.AUTH_PASSWORD
            and const_eq(user, config.AUTH_USER)
            and const_eq(password, config.AUTH_PASSWORD)
        ):
            role = "operator"
        elif (
            config.VIEWER_USER
            and config.VIEWER_PASSWORD
            and const_eq(user, config.VIEWER_USER)
            and const_eq(password, config.VIEWER_PASSWORD)
        ):
            role = "viewer"
        if role:
            session.clear()
            session["authenticated"] = True
            session["username"] = user
            session["role"] = role
            session.permanent = True
            app.permanent_session_lifetime = datetime.timedelta(hours=8)
            nxt = request.args.get("next") or url_for("index")
            if not nxt.startswith("/"):
                nxt = url_for("index")
            return redirect(nxt)
        error = "Invalid credentials"
        time.sleep(0.4)
    return render_template("login.html", company=COMPANY, error=error)
@app.route("/logout")
def logout():
    session.clear()
    return redirect(url_for("login") if config.AUTH_REQUIRED else url_for("index"))
