def ensure_sample_reports():
    if not config.DEMO_MODE:
        return
    for fname in [
        "CyberAudit_Report_AUD-20260829-001.pdf",
        "CyberAudit_Report_AUD-20260828-003.pdf",
    ]:
        path = config.REPORTS_DIR / fname
        if not path.exists():
            path.write_bytes(b"%PDF-1.4 placeholder CyberAudit report\n")

def ensure_sample_playbooks():
    samples = {
        "PB-IR-001_Containment_HighSeverity.md": "# IR High Severity\nIsolate host, notify SOC, map CMA ss.3/6/7.\n",
        "PB-EXFIL-002_DataExfiltration.md": "# Data Exfiltration\nQuarantine source, notify DPO, capture evidence.\n",
        "PB-COMP-003_CMA_Section6.md": "# CMA Section 6\nTag unauthorized access findings to s.6.\n",
    }
    for name, body in samples.items():
        p = config.PLAYBOOKS_DIR / name
        if not p.exists():
            p.write_text(body, encoding="utf-8")
