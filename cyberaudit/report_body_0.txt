def _plain_for(service: str, title: str) -> dict:
    key = (service or "").lower()
    for k, v in PLAIN.items():
        if k in key or k in (title or "").lower():
            return v
    return PLAIN["default"]


def severity_label(sev: str) -> str:
    return {
        "critical": "Critical — address immediately",
        "high": "High — plan remediation this week",
        "medium": "Medium — schedule within a reasonable window",
        "low": "Low — improve when convenient",
        "info": "Informational",
    }.get((sev or "").lower(), sev or "Unknown")


def risk_score(vulns: list[dict]) -> tuple[float, str]:
    crit = sum(1 for v in vulns if v.get("severity") == "critical")
    high = sum(1 for v in vulns if v.get("severity") == "high")
    med = sum(1 for v in vulns if v.get("severity") == "medium")
    low = sum(1 for v in vulns if v.get("severity") == "low")
    score = max(0.0, 100.0 - crit * 25 - high * 12 - med * 5 - low * 1)
    if score >= 85:
        band = "Good — limited issues found"
    elif score >= 70:
        band = "Fair — several items need attention"
    elif score >= 50:
        band = "Needs improvement — prioritise high findings"
    else:
        band = "Poor — urgent remediation recommended"
    return score, band


def build_executive_summary(
    session_id: str,
    target: str,
    open_ports: list[dict],
    vulns: list[dict],
    hosts_up: list[str],
) -> str:
    score, band = risk_score(vulns)
    crit = sum(1 for v in vulns if v.get("severity") == "critical")
    high = sum(1 for v in vulns if v.get("severity") == "high")
    med = sum(1 for v in vulns if v.get("severity") == "medium")
    low = sum(1 for v in vulns if v.get("severity") == "low")

    lines = [
        "=" * 72,
        f"  {COMPANY['name']}".center(72),
        f"  Network Security Audit — Executive Summary".center(72),
        f"  {COMPANY['tagline']}".center(72),
        "=" * 72,
        "",
        f"Report ID:     {session_id}",
        f"Date:          {datetime.datetime.now().strftime('%d %B %Y, %H:%M')}",
        f"Scope:         {target}",
        f"Contact:       {COMPANY['email']} | {COMPANY['phone']}",
        f"Website:       {COMPANY['website']}",
        "",
        "-" * 72,
        "WHAT THIS REPORT IS",
        "-" * 72,
        "This is a point-in-time network audit. It maps devices and services",
        "that responded during the scan window. It is not continuous 24/7",
        "monitoring. Results reflect what was reachable from the scanner",
        "at the time of the engagement, under the agreed rules of engagement.",
        "",
        "-" * 72,
        "HEADLINE FOR LEADERSHIP",
        "-" * 72,
        f"Overall posture score:  {score:.0f} / 100  ({band})",
        f"Systems responding:     {len(hosts_up)}",
        f"Open network services:  {len(open_ports)}",
        f"Issues logged:          {len(vulns)}  "
        f"(Critical {crit} · High {high} · Medium {med} · Low {low})",
        "",
    ]

    if crit or high:
        lines.append("Top priority for management:")
        lines.append("  One or more high-impact issues were found. These should be")
        lines.append("  assigned to IT or your managed service provider without delay.")
        lines.append("")
    elif not vulns:
        lines.append("No major service-exposure issues were flagged by this scan.")
        lines.append("Continue normal patching, backups, and access reviews.")
        lines.append("")

    lines.extend(["-" * 72, "SYSTEMS SEEN ON THE NETWORK", "-" * 72])
    if hosts_up:
        for h in hosts_up:
            ports_here = [p for p in open_ports if p.get("host") == h]
            svc = ", ".join(f"{p.get('port')}/{p.get('service')}" for p in ports_here[:8]) or "no common ports open"
            lines.append(f"  • {h}")
            lines.append(f"      Services: {svc}")
    else:
        lines.append("  No hosts responded in scope (check connectivity / firewall).")

    lines.extend(["", "-" * 72, "FINDINGS IN PLAIN LANGUAGE", "-" * 72, ""])

    if not vulns:
        lines.append("  No risk findings were generated from open services in scope.")
    else:
        order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
        sorted_v = sorted(vulns, key=lambda v: order.get(v.get("severity") or "info", 9))
        for i, v in enumerate(sorted_v, 1):
            plain = _plain_for(v.get("service") or "", v.get("title") or "")
            lines.append(f"  [{i}] {severity_label(v.get('severity') or '')}")
            lines.append(f"      Where:   {v.get('host')}:{v.get('port')} ({v.get('service') or 'service'})")
            lines.append(f"      What:    {plain['what']}")
            lines.append(f"      Why it matters: {plain['why']}")
            lines.append(f"      What to do:     {plain['do']}")
            if v.get("cma_section"):
                lines.append(
                    f"      Local law note: Related to Trinidad & Tobago "
                    f"Computer Misuse Act — {v.get('cma_section')} "
                    f"(unauthorised access / system interference themes)."
                )
            lines.append("")

    lines.extend([
        "-" * 72,
        "RECOMMENDED NEXT STEPS",
        "-" * 72,
        "  1. Assign an owner for each Critical and High finding.",
        "  2. Close or restrict unnecessary services at the firewall.",
        "  3. Ensure backups are tested and offline/immutable copies exist.",
        "  4. Review who has remote access (VPN, RDP, SSH) and enforce MFA.",
        "  5. Schedule a follow-up scan after remediation to confirm closure.",
        "",
        "-" * 72,
        "REGULATORY CONTEXT (TRINIDAD & TOBAGO)",
        "-" * 72,
        "  This engagement references the Computer Misuse Act, Chap. 11:17",
        "  (Sections 3, 6 and 7) for awareness. Mapping is indicative for",
        "  risk discussion — not a formal legal opinion.",
        "",
        "-" * 72,
        "DISCLAIMER",
        "-" * 72,
        "  This report is confidential to the client named in the engagement.",
        "  Scanning was performed only within authorised scope. Absence of a",
        "  finding does not guarantee absence of risk. For questions contact",
        f"  {COMPANY['email']} or {COMPANY['phone']}.",
        "",
        f"  Prepared by {COMPANY['name']} — {COMPANY['address']}",
        "=" * 72,
        "",
    ])
    return "\n".join(lines)
