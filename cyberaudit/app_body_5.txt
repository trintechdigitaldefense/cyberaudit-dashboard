    if not name:
        return jsonify({"error": "module_name required"}), 400
    row = upsert_module_status(
        module_name=name,
        status=data.get("status") or "idle",
        last_result=data.get("last_result") or data.get("result") or "",
        version=data.get("version") or "",
        last_run=data.get("last_run"),
    )
    return jsonify({"ok": True, "module": row}), 201

@app.route("/api/events/<int:event_id>/ack", methods=["POST"])
@login_required
@operator_required
def ack_event(event_id: int):
    who = session.get("username") or "operator"
    now = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with get_db() as conn:
        cur = conn.execute(
            "UPDATE live_events SET acknowledged=1, acknowledged_by=?, acknowledged_at=? WHERE id=?",
            (who, now, event_id),
        )
        conn.commit()
        if cur.rowcount == 0:
            return jsonify({"error": "not_found"}), 404
    return jsonify({"ok": True, "id": event_id, "acknowledged_by": who})
