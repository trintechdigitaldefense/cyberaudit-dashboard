def _now() -> str:
    return datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")


def _mod(name: str, status: str, result: str = "") -> None:
    upsert_module_status(module_name=name, status=status, last_result=result, last_run=_now())


def _ev(module: str, message: str, severity: str = "info", **kw: Any) -> None:
    insert_live_event(
        module=module,
        message=message,
        severity=severity,
        event_type=kw.get("event_type")
        or ("alert" if severity in ("high", "critical") else "info"),
        details=kw.get("details"),
    )


def _parse_targets(target: str, max_hosts: int = 256) -> list[str]:
    target = (target or "").strip()
    if not target:
        return []
    hosts: list[str] = []
    for part in [p.strip() for p in target.replace(";", ",").split(",") if p.strip()]:
        try:
            net = ipaddress.ip_network(part, strict=False)
            all_hosts = list(net.hosts())
            if len(all_hosts) > max_hosts:
                step = max(1, len(all_hosts) // max_hosts)
                all_hosts = all_hosts[::step][:max_hosts]
            hosts.extend(str(h) for h in all_hosts)
            continue
        except ValueError:
            pass
        try:
            ipaddress.ip_address(part)
            hosts.append(part)
            continue
        except ValueError:
            try:
                infos = socket.getaddrinfo(part, None, socket.AF_INET)
                for info in infos:
                    hosts.append(info[4][0])
            except Exception:
                hosts.append(part)
    seen, out = set(), []
    for h in hosts:
        if h not in seen:
            seen.add(h)
            out.append(h)
    return out[:max_hosts]


def _host_alive(host: str, timeout: float = 0.8) -> bool:
    for port in (80, 443, 22, 445, 3389, 135):
        try:
            with socket.create_connection((host, port), timeout=timeout):
                return True
        except Exception:
            continue
    try:
        r = subprocess.run(["ping", "-c", "1", "-W", "1", host], capture_output=True, timeout=3)
        if r.returncode == 0:
            return True
    except Exception:
        pass
    return False


def _tcp_probe(host: str, port: int, timeout: float = 1.0) -> Optional[dict]:
    try:
        with socket.create_connection((host, port), timeout=timeout) as s:
            banner = ""
            try:
                s.settimeout(0.6)
                if port in (21, 22, 25, 110, 143):
                    banner = s.recv(256).decode("utf-8", errors="replace").strip()[:180]
            except Exception:
                pass
            return {
                "host": host,
                "port": str(port),
                "service": SERVICE_HINTS.get(port, "unknown"),
                "banner": banner,
                "state": "open",
            }
    except Exception:
        return None


def _nmap_available() -> bool:
    try:
        return subprocess.run(["nmap", "-V"], capture_output=True, text=True, timeout=5).returncode == 0
    except Exception:
        return False


def discover_live_hosts(hosts: list[str]) -> list[str]:
    _mod("nmap_module", "running", f"Host discovery on {len(hosts)} address(es)")
    _ev("nmap_module", f"Discovering live systems ({len(hosts)} candidates)")
    live: list[str] = []
    with ThreadPoolExecutor(max_workers=64) as pool:
        futs = {pool.submit(_host_alive, h): h for h in hosts}
        for fut in as_completed(futs):
            if fut.result():
                live.append(futs[fut])
    if not live and len(hosts) == 1:
        live = list(hosts)
    live = sorted(set(live), key=lambda x: tuple(int(p) if p.isdigit() else 0 for p in x.replace(":", ".").split(".")[:4]))
    _ev("nmap_module", f"Live systems: {len(live)}", details={"hosts": live[:50]})
    return live
