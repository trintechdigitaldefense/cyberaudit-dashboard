@app.route("/api/findings/<int:fid>/status", methods=["POST"])
@login_required
@operator_required
def update_finding_status(fid: int):
    data = request.get_json(silent=True) or {}
    status = (data.get("status") or "").strip().lower()
    if status not in ("open", "in_remediation", "mitigated", "closed", "accepted", "false_positive", "new"):
        return jsonify({"error": "invalid status"}), 400
    with get_db() as conn:
        cur = conn.execute("UPDATE findings SET status=? WHERE id=?", (status, fid))
        conn.commit()
        if cur.rowcount == 0:
            return jsonify({"error": "not_found"}), 404
    return jsonify({"ok": True, "id": fid, "status": status})

@socketio.on("connect")
def on_connect():
    if not socket_authenticated():
        disconnect()
        return False
    from flask import request as _req
    from flask import session as _sess
    from cyberaudit.presence import record_connect, snapshot as presence_snapshot
    username = (_sess.get("user") or _sess.get("username") or "operator")
    role = (_sess.get("role") or "operator")
    rec = record_connect(
        sid=getattr(_req, "sid", "") or "",
        username=str(username),
        role=str(role),
        ip=(_req.headers.get("X-Forwarded-For") or _req.remote_addr or "")[:64],
        user_agent=(_req.headers.get("User-Agent") or "")[:120],
    )
    emit("status", {"msg": "Connected to CyberAudit live feed", "time": rec.get("connected_at"), "username": rec.get("username")})
    emit("presence", presence_snapshot())
    socketio.emit("presence", presence_snapshot())

@socketio.on("disconnect")
def on_disconnect():
    from flask import request as _req
    from cyberaudit.presence import record_disconnect, snapshot as presence_snapshot
    sid = getattr(_req, "sid", "") or ""
    record_disconnect(sid)
    try:
        socketio.emit("presence", presence_snapshot())
    except Exception:
        pass

_workers_started = False
def start_background_workers():
    init_db()
    ensure_sample_playbooks()
    from cyberaudit.modules_bootstrap import ensure_modules_ready
    ensure_modules_ready()
    from cyberaudit.workers import job_runner
    threading.Thread(target=job_runner, daemon=True).start()
    threading.Thread(target=inbox_poller, daemon=True).start()
    threading.Thread(target=retention_worker, daemon=True).start()
    if config.DEMO_MODE:
        if db_is_empty():
            seed_sample_data()
        ensure_sample_reports()
        threading.Thread(target=live_feed_simulator, daemon=True).start()
        logger.warning("DEMO_MODE=1 — synthetic feed ON")
    else:
        logger.info("REAL MODE — job runner active")
def ensure_workers():
    global _workers_started
    if _workers_started:
        return
    _workers_started = True
    start_background_workers()

@app.route("/api/dashboard")
@login_required
def api_dashboard():
    from cyberaudit.dashboard_features import system_snapshot
    snap = system_snapshot()
    with get_db() as conn:
        modules = [dict(r) for r in conn.execute("SELECT * FROM module_status").fetchall()]
        sessions = [dict(r) for r in conn.execute("SELECT * FROM audit_sessions ORDER BY start_time DESC LIMIT 15").fetchall()]
        findings_n = conn.execute("SELECT COUNT(*) FROM findings").fetchone()[0]
        try:
            high_n = conn.execute("SELECT COUNT(*) FROM findings WHERE lower(severity) IN ('high','critical')").fetchone()[0]
        except Exception:
            high_n = 0
    snap["modules"] = modules
    snap["sessions"] = sessions
    snap["findings_count"] = findings_n
    snap["high_critical_count"] = high_n
    snap["role"] = current_role()
    snap["username"] = session.get("username")
    return jsonify(snap)

@app.route("/api/v1/jobs", methods=["POST"])
def api_v1_jobs_create():
    from flask import request as _req
    from cyberaudit.partner_api import verify_api_key, enqueue_partner_job
    provided = _req.headers.get("X-API-Key", "")
    if not provided:
        auth = _req.headers.get("Authorization", "")
        if auth.lower().startswith("bearer "):
            provided = auth[7:].strip()
    partner = verify_api_key(provided)
    if not partner:
        return jsonify({"error": "unauthorized", "api_version": "v1"}), 401
    payload = _req.get_json(silent=True) or {}
    try:
        result = enqueue_partner_job(payload, partner)
        return jsonify(result), 202
    except ValueError as ve:
        return jsonify({"error": "validation_error", "detail": str(ve)}), 400
    except Exception as exc:
        logger.exception("partner job")
        return jsonify({"error": "server_error", "detail": str(exc)[:200]}), 500
