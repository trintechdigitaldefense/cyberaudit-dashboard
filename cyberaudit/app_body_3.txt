@app.route("/api/playbooks/<path:name>/trigger", methods=["POST"])
@login_required
@operator_required
def trigger_playbook(name: str):
    path = safe_file_under(config.PLAYBOOKS_DIR, name)
    if path is None:
        return jsonify({"error": "playbook not found"}), 404
    data = request.get_json(silent=True) or {}
    host = (data.get("host") or data.get("target") or "").strip()
    who = session.get("username") or "operator"
    now = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    job_id = f"PB-{datetime.datetime.now().strftime('%Y%m%d-%H%M%S')}-{secrets.token_hex(2)}"
    payload = {
        "type": "playbook_trigger",
        "job_id": job_id,
        "playbook": path.name,
        "host": host,
        "requested_by": who,
        "created_at": now,
        "notes": data.get("notes") or "",
    }
    out = config.JOBS_DIR / f"{job_id}.json"
    out.write_text(json.dumps(payload, indent=2), encoding="utf-8")
    insert_live_event(
        module="incident_response_module",
        message=f"Playbook triggered: {path.name}" + (f" host={host}" if host else ""),
        severity="high",
        event_type="playbook",
        details=payload,
    )
    return jsonify({"ok": True, "job": payload, "job_file": str(out)}), 201
@app.route("/api/findings/<int:fid>/status", methods=["POST"])
@login_required
@operator_required
def update_finding_status(fid: int):
    data = request.get_json(silent=True) or {}
    status = (data.get("status") or "").strip().lower()
    if status not in ("open", "mitigated", "accepted", "false_positive"):
        return jsonify({"error": "invalid status"}), 400
    with get_db() as conn:
        cur = conn.execute("UPDATE findings SET status=? WHERE id=?", (status, fid))
        conn.commit()
        if cur.rowcount == 0:
            return jsonify({"error": "not_found"}), 404
    return jsonify({"ok": True, "id": fid, "status": status})
@socketio.on("connect")
def on_connect():
    if not socket_authenticated():
        disconnect()
        return False
    emit(
        "status",
        {
            "msg": "Connected to CyberAudit live feed",
            "time": datetime.datetime.now().isoformat(),
        },
    )
@socketio.on("request_snapshot")
def on_snapshot():
    if not socket_authenticated():
        disconnect()
        return
    with get_db() as conn:
        events = [
            dict(r)
            for r in conn.execute(
                "SELECT * FROM live_events ORDER BY id DESC LIMIT 25"
            ).fetchall()
        ]
    emit("snapshot", events)
_workers_started = False
def start_background_workers():
    init_db()

    if config.DEMO_MODE:
        if db_is_empty():
            seed_sample_data()

        threading.Thread(target=live_feed_simulator, daemon=True).start()
        logger.warning("DEMO_MODE=1 — simulator active")

    threading.Thread(target=retention_worker, daemon=True).start()
def ensure_workers():
    global _workers_started
    if _workers_started:
        return
    _workers_started = True
    start_background_workers()
