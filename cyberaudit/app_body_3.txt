@login_required
def api_events():
    limit = min(request.args.get("limit", 50, type=int), 500)
    with get_db() as conn:
        rows = [
            dict(r)
            for r in conn.execute(
                "SELECT * FROM live_events ORDER BY id DESC LIMIT ?", (limit,)
            ).fetchall()
        ]
    return jsonify(rows)
